cmake_minimum_required(VERSION 3.22)
project(openweights_session_cache_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(ENGINE "${CMAKE_CURRENT_SOURCE_DIR}/../../main/cpp")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
option(OW_CACHE_SANITIZERS "Check reply offsets and ownership with ASan and UBSan" OFF)
if(OW_CACHE_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
include("${ENGINE}/PatchGGMLGraphOffset.cmake")
add_subdirectory("${ENGINE}/llama.cpp" llama EXCLUDE_FROM_ALL)
add_executable(session-cache-tests session_cache_test.cpp "${ENGINE}/engine_session.cpp")
target_include_directories(session-cache-tests PRIVATE "${ENGINE}"
    "${ENGINE}/llama.cpp/common" "${ENGINE}/llama.cpp/tools/mtmd" "${ENGINE}/llama.cpp/vendor")
target_link_libraries(session-cache-tests PRIVATE llama llama-common mtmd)
set(OW_CACHE_TEST_MODEL "" CACHE FILEPATH "Pinned Qwen3-0.6B-Q4_K_M.gguf, kept outside source control")
if(OW_CACHE_TEST_MODEL)
    enable_testing()
    add_test(NAME session-cache-regressions COMMAND session-cache-tests "${OW_CACHE_TEST_MODEL}")
endif()
